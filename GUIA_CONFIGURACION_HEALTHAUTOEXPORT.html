1 Ejecutar Migración de Base de Datos

Primero, ejecuta estos comandos SQL en tu base de datos PostgreSQL:

-- Agregar nuevos campos a DATOS_VITALES ALTER TABLE DATOS_VITALES ADD COLUMN FECHA_MEDICION TIMESTAMP; ALTER TABLE DATOS_VITALES ADD COLUMN DISPOSITIVO VARCHAR(100); ALTER TABLE DATOS_VITALES ADD COLUMN FUENTE VARCHAR(100); -- Actualizar registros existentes UPDATE DATOS_VITALES SET FECHA_MEDICION = FECHA_REGISTRO WHERE FECHA_MEDICION IS NULL; -- Crear tabla de configuración CREATE TABLE HEALTH_AUTO_EXPORT_CONFIGS ( ID SERIAL PRIMARY KEY, USUARIO_ID INTEGER NOT NULL, API_KEY VARCHAR(100) NOT NULL, IS_ACTIVE BOOLEAN NOT NULL DEFAULT TRUE, CREATED_AT TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP, LAST_SYNC_AT TIMESTAMP NULL, ALLOWED_DATA_TYPES VARCHAR(1000) NULL, DEVICE_INFO VARCHAR(500) NULL, SYNC_INTERVAL_MINUTES INTEGER NOT NULL DEFAULT 60, CONSTRAINT FK_HEALTH_AUTO_EXPORT_CONFIGS_USUARIOS FOREIGN KEY (USUARIO_ID) REFERENCES USUARIOS(ID) ON DELETE CASCADE ); -- Crear índices CREATE UNIQUE INDEX IX_HEALTH_AUTO_EXPORT_CONFIGS_API_KEY ON HEALTH_AUTO_EXPORT_CONFIGS (API_KEY);
